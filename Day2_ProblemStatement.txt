// Find Minimum Order Amount
db.orders.aggregate([
  { $group: { _id: null, minAmount: { $min: "$totalAmount" } } }
])

// Find Maximum Order Amount
db.orders.aggregate([
  { $group: { _id: null, maxAmount: { $max: "$totalAmount" } } }
])

// Find Average Order Amount
db.orders.aggregate([
  { $group: { _id: null, avgAmount: { $avg: "$totalAmount" } } }
])

// Find Total Number of Orders (Count)
db.orders.aggregate([
  { $group: { _id: null, totalOrders: { $sum: 1 } } }
])

// Find Total Revenue (Sum)
db.orders.aggregate([
  { $group: { _id: null, totalRevenue: { $sum: "$totalAmount" } } }
])

// Filter Orders > 10000
db.orders.aggregate([
  { $match: { totalAmount: { $gt: 10000 } } }
])

// Show Only Customer & Amount
db.orders.aggregate([
  { $project: { customerName: 1, totalAmount: 1, _id: 0 } }
])

// Limit Results
db.orders.aggregate([
  { $limit: 5 }
])

// Skip First 3 Orders
db.orders.aggregate([
  { $skip: 3 }
])

// Sort by Total Amount Descending
db.orders.aggregate([
  { $sort: { totalAmount: -1 } }
])

// Group by Category with Total Revenue
db.orders.aggregate([
  {
    $group: {
      _id: "$category",
      totalRevenue: { $sum: "$totalAmount" },
      avgAmount: { $avg: "$totalAmount" },
      maxAmount: { $max: "$totalAmount" },
      minAmount: { $min: "$totalAmount" },
      count: { $sum: 1 }
    }
  }
])

// Combine $skip + $limit (Pagination) - Skip 5, Show Next 5
db.orders.aggregate([
  { $skip: 5 },
  { $limit: 5 }
])

// Sort by totalAmount descending, skip first 2, show next 3
db.orders.aggregate([
  { $sort: { totalAmount: -1 } },
  { $skip: 2 },
  { $limit: 3 }
])

// Find total revenue for each category where order amount is greater than 10,000, show only category and revenue, sort descending
db.orders.aggregate([
  { $match: { totalAmount: { $gt: 10000 } } },
  { $group: { _id: "$category", totalRevenue: { $sum: "$totalAmount" } } },
  { $project: { category: "$_id", totalRevenue: 1, _id: 0 } },
  { $sort: { totalRevenue: -1 } }
])

// Find average order amount per product, show top 3 products with highest average
db.orders.aggregate([

  { $project: { product: "$_id", avgAmount: 1, _id: 0 } },
  { $sort: { avgAmount: -1 } },
  { $limit: 3 }
])
