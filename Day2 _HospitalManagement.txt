//Treatment cost > 20,000, show name & department 
db.patients.aggregate([
  { $match: { treatmentCost: { $gt: 20000 } } },
  { $project: { _id: 0, patientName: 1, department: 1 } }
])


//Sort patients by treatment cost (descending) 
db.patients.aggregate([
  { $sort: { treatmentCost: -1 } }
])


//Display first 5 patient records 
db.patients.aggregate([
  { $limit: 5 }
])


//Skip first 3 patients 
db.patients.aggregate([
  { $skip: 3 }
])


//Cardiology patients, show only names 
db.patients.aggregate([
  { $match: { department: "Cardiology" } },
  { $project: { _id: 0, patientName: 1 } }
])


//Total treatment cost per department 
db.patients.aggregate([
  {$group: {_id: "$department", totalTreatmentCost: { $sum: "$treatmentCost" } }  }
])


//Average treatment cost per doctor (ascending order) 
db.patients.aggregate([
  { $group: {
      _id: "$doctor",
      avgTreatmentCost: { $avg: "$treatmentCost" }} },
  { $sort: { avgTreatmentCost: 1 } }
])


//Departments with total treatment cost > 50,000 
db.patients.aggregate([
  {
    $group: {
      _id: "$department",
      totalCost: { $sum: "$treatmentCost" }  } },
  {
    $match: {
      totalCost: { $gt: 50000 }   } }
])


//Number of patients in each department (descending) 
db.patients.aggregate([
  {
    $group: {
      _id: "$department",
      patientCount: { $sum: 1 }}  },
  { $sort: { patientCount: -1 } }
])


//Top 3 doctors with highest treatment cost 
db.patients.aggregate([
  {
    $group: {
      _id: "$doctor",
      totalTreatmentCost: { $sum: "$treatmentCost" }} },
  { $sort: { totalTreatmentCost: -1 } },
  { $limit: 3 }
])


//Age > 40, group by department and count patients 
db.patients.aggregate([
  { $match: { age: { $gt: 40 } } },
  {
    $group: {
      _id: "$department",
      patientCount: { $sum: 1 } }
])
//Average treatment cost per department, show department & avg cost, sort descending 
db.patients.aggregate([
  {
    $group: {
      _id: "$department",
      avgCost: { $avg: "$treatmentCost" }}  },
  {
    $project: {
      _id: 0,
      department: "$_id",
      avgCost: 1} },
  { $sort: { avgCost: -1 } }
])




//Departments having more than 2 patients 
db.patients.aggregate([
  {
    $group: {
      _id: "$department",
      patientCount: { $sum: 1 }}  },
  { $match: { patientCount: { $gt: 2 } } }
])


//Top 3 departments by total treatment cost after skipping first one 
db.patients.aggregate([
  { $group: {
      _id: "$department",
      totalCost: { $sum: "$treatmentCost" }} },
  { $sort: { totalCost: -1 } },
  { $skip: 1 },
  { $limit: 3 }
])


//Doctors treating patients with cost > 10,000, total cost, top 2 
db.patients.aggregate([
  { $match: { treatmentCost: { $gt: 10000 } } },
  {$group: {
      _id: "$doctor",
      totalCost: { $sum: "$treatmentCost" }  }  },
  { $sort: { totalCost: -1 } },
  { $limit: 2 }
])
